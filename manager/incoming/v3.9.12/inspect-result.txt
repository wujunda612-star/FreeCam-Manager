function Replace-One([string]$path, [string]$old, [string]$new, [string]$label) {
    $text = [System.IO.File]::ReadAllText($path)
    $first = $text.IndexOf($old, [StringComparison]::Ordinal)
    if ($first -lt 0) { throw "${label}: marker not found in $path" }
    if ($text.IndexOf($old, $first + $old.Length, [StringComparison]::Ordinal) -ge 0) { throw "${label}: marker is not unique in $path" }
    Write-Utf8 $path ($text.Substring(0,$first) + $new + $text.Substring($first + $old.Length))
}
function Replace-RegexOne([string]$path, [string]$pattern, [string]$new, [string]$label) {
    $text = [System.IO.File]::ReadAllText($path)
    $matches = [regex]::Matches($text, $pattern, [System.Text.RegularExpressions.RegexOptions]::Singleline)
    if ($matches.Count -ne 1) { throw "${label}: expected exactly 1 match in $path, found $($matches.Count)" }
    $m = $matches[0]
    Write-Utf8 $path ($text.Substring(0,$m.Index) + $new + $text.Substring($m.Index + $m.Length))
}
